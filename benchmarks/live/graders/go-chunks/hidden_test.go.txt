package chunks

import (
	"reflect"
	"slices"
	"testing"
)

func TestNormalize(t *testing.T) {
	for _, size := range []int{-3, 0, 1, 2, 9} {
		if got := NormalizeChunkSize(size); got != max(1, size) {
			t.Errorf("NormalizeChunkSize(%d)=%d", size, got)
		}
	}
}

func TestSplit(t *testing.T) {
	for _, test := range []struct {
		values []int
		size   int
		want   [][]int
	}{
		{nil, 2, nil}, {[]int{}, 0, nil}, {[]int{1}, 2, [][]int{{1}}},
		{[]int{1, 2, 3, 4, 5}, 2, [][]int{{1, 2}, {3, 4}, {5}}},
		{[]int{1, 2, 3, 4}, 2, [][]int{{1, 2}, {3, 4}}},
		{[]int{1, 2}, 0, [][]int{{1}, {2}}},
		{[]int{1, 2}, -3, [][]int{{1}, {2}}},
		{[]int{1, 2}, 9, [][]int{{1, 2}}},
	} {
		before := slices.Clone(test.values)
		if got := Split(test.values, test.size); !reflect.DeepEqual(got, test.want) {
			t.Errorf("Split(%v,%d)=%v; want %v", test.values, test.size, got, test.want)
		}
		if !slices.Equal(before, test.values) {
			t.Fatal("input was mutated")
		}
	}
}
