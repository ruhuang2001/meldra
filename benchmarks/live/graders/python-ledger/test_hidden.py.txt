import unittest
from money import parse_cents
from ledger import total_cents


class LedgerTests(unittest.TestCase):
    def test_exact_amounts(self):
        for text, expected in [("1.15", 115), ("0.29", 29), ("12", 1200),
                               ("-0.05", -5), ("+3.4", 340), ("0001.20", 120)]:
            with self.subTest(text=text):
                self.assertEqual(parse_cents(text), expected)

    def test_invalid_amounts(self):
        for text in ["", "1.234", "1.", ".5", " 1.0", "1e2", "１２", "nan", "--1"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_cents(text)

    def test_total(self):
        self.assertEqual(total_cents([]), 0)
        self.assertEqual(total_cents([{"amount": "1.15"}, {"amount": "-0.05"},
                                     {"amount": "99.99", "void": True}]), 110)
        self.assertEqual(total_cents([{"amount": "bad", "void": True}]), 0)
