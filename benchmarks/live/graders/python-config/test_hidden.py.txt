import copy
import unittest
from config import merge_config


class ConfigTests(unittest.TestCase):
    def test_merge_and_independence(self):
        defaults = {"host": "local", "db": {"port": 5, "options": [1]}, "tags": ["a"]}
        overrides = {"db": {"port": 6, "ssl": True}, "tags": ["b"], "new": {"list": [3]}}
        before = copy.deepcopy((defaults, overrides))
        result = merge_config(defaults, overrides)
        self.assertEqual(result, {"host": "local", "db": {"port": 6, "options": [1], "ssl": True},
                                  "tags": ["b"], "new": {"list": [3]}})
        result["db"]["options"].append(9)
        result["tags"].append("c")
        result["new"]["list"].append(8)
        self.assertEqual((defaults, overrides), before)

    def test_type_changes(self):
        self.assertEqual(merge_config({"a": {"b": 1}}, {"a": None}), {"a": None})
        self.assertEqual(merge_config({"a": 1}, {"a": {"b": 2}}), {"a": {"b": 2}})
        self.assertEqual(merge_config({}, {}), {})
